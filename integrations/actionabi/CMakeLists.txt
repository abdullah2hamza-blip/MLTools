cmake_minimum_required(VERSION 3.26)
project(MLToolsActionABIControls LANGUAGES CXX)

if(NOT IS_DIRECTORY "${ACTIONABI_SOURCE_DIR}")
  message(FATAL_ERROR "Pass the verified complete ActionABI checkout")
endif()
set(ACTIONABI_ENABLE_CUDA OFF CACHE BOOL "" FORCE)
set(ACTIONABI_ENABLE_PINOCCHIO OFF CACHE BOOL "" FORCE)
set(ACTIONABI_BUILD_PYBIND OFF CACHE BOOL "" FORCE)
set(ACTIONABI_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(ACTIONABI_BUILD_TESTS ON CACHE BOOL "" FORCE)
enable_testing()
add_subdirectory("${ACTIONABI_SOURCE_DIR}" actionabi-upstream)
add_executable(mltools_actionabi_controls controls.cpp)
target_link_libraries(mltools_actionabi_controls PRIVATE actionabi_core)
target_compile_features(mltools_actionabi_controls PRIVATE cxx_std_20)
